class fifo_gen;
  mailbox gen2drv;
  int num_transactions;
  event done;
  
  fifo_trans gen_trans;

  function new(mailbox gen2drv);
    this.gen2drv = gen2drv;
  endfunction

  task run();
    for (int i = 0; i < num_transactions; i++) begin
      fifo_trans tr;
      if (gen_trans == null) tr = new();
      else $cast(tr, gen_trans.new()); 
      
      if (!tr.randomize()) $fatal("[GENERATOR] Randomization configuration failure!");
      gen2drv.put(tr);
    end
    -> done;
  endtask
endclass
